  if timeout "$PROBE_TIMEOUT_SECONDS" ssh -o BatchMode=yes -o ConnectTimeout="$SSH_CONNECT_TIMEOUT_SECONDS" "$HOST" "test -f '$REMOTE_STATUS' && test ! -L '$REMOTE_STATUS' && grep -q '^finished_at=' '$REMOTE_STATUS'"; then

  else
    PROBE_STATUS=$?
    printf 'observed_at=%s\npoll=%s\nterminal_probe_exit=%s\n' "$(date -Iseconds)" "$POLL_COUNT" "$PROBE_STATUS"
  fi
  sleep "$POLL_SECONDS"
done
