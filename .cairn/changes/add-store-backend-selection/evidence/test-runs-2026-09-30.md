## Run 28: reproducible, nonsecret test-only signer provisioning

Run 25's fixture signer is **not** merely an ephemeral scratch-file path.
Its deliberately nonsecret Ed25519 test seed is 32 repetitions of byte
`0x2a`; Run 27's alternate fixture seed is 32 repetitions of `0x2b`.

be reused for production signing. The following command, run in the pinned
isolated worktree's Nix development shell, derives both public keys with
OpenSSL, provisions only their Nix-format **test-only** key files under
`/home/brittonr/scratch`, refuses to overwrite a mismatched existing file,
and prints **only** public tokens and SHA-256 fingerprints. No private
keypair token appears in stdout or retained Cairn evidence.

```sh
nix develop --offline --no-write-lock-file --command python3 - <<'PY'
import base64
import hashlib
import os
import pathlib
import stat
import subprocess

scratch = pathlib.Path("/home/brittonr/scratch")
assert scratch.is_dir()
fixtures = (
    (0x2a, "mantle-baseline-fixture-1", "mantle-baseline-signing-key",
     "GX9rI+FshTLGq8g4+s1ep4m+DHaykgM0A5v6iz02jWE=",
     "95931c8ccd73b6ce3e0f5d6f5fdbfeffb4a1a4c459699cd9b522626afbbf56fe"),
    (0x2b, "mantle-secondary-fixture-1", "mantle-secondary-signing-key",
     "RQigeqlBcH8+stuUyIl6gLLBGXR2tt4hOsJz332GxP8=",
     "c2403ab3b0edf3f11c5c9097e831df1ac6c7b6d32c4bb93faed7a0d6d5c2dc87"),
)
for byte, name, filename, expected_public, expected_file_sha256 in fixtures:
    seed = bytes([byte]) * 32
    private_der = bytes.fromhex("302e020100300506032b657004220420") + seed
    public_der = subprocess.run(
        ["openssl", "pkey", "-inform", "DER", "-pubout", "-outform", "DER"],
        input=private_der, capture_output=True, check=True,
    ).stdout
    public = public_der[-32:]
    public_token = base64.b64encode(public).decode("ascii")
    assert public_token == expected_public
    fixture_bytes = (name + ":" + base64.b64encode(seed + public).decode("ascii") + "\n").encode()
    assert hashlib.sha256(fixture_bytes).hexdigest() == expected_file_sha256
    path = scratch / filename
    if path.exists():
        assert path.read_bytes() == fixture_bytes, "never overwrite existing signing material"
    else:
        fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
        with os.fdopen(fd, "wb") as output:
            output.write(fixture_bytes)
    assert stat.S_IMODE(path.stat().st_mode) == 0o600
    print(name + ":" + public_token, expected_file_sha256)
PY
```
