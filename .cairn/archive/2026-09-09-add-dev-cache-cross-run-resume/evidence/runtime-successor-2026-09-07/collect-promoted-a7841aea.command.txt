copy_observation() {
  RELATIVE=$1
  BYTES_MAX=$2
  if ssh -o BatchMode=yes -o ConnectTimeout="$SSH_CONNECT_TIMEOUT_SECONDS" "$HOST" "test -f '$STAGING/$RELATIVE' && test ! -L '$STAGING/$RELATIVE'"; then
    ssh -o BatchMode=yes -o ConnectTimeout="$SSH_CONNECT_TIMEOUT_SECONDS" "$HOST" "test \"\$(stat -c %s '$STAGING/$RELATIVE')\" -le '$BYTES_MAX'"
    mkdir -p "$OUTPUT/$(dirname "$RELATIVE")"
    scp -q -p -o BatchMode=yes -o ConnectTimeout="$SSH_CONNECT_TIMEOUT_SECONDS" "$HOST:$STAGING/$RELATIVE" "$OUTPUT/$RELATIVE"
    printf 'copied %s\n' "$RELATIVE" >> "$OUTPUT/observations.txt"
  else
    OBSERVATION_STATUS=$?
    EXPECTED_ABSENCE_STATUS=1
    test "$OBSERVATION_STATUS" -eq "$EXPECTED_ABSENCE_STATUS"
    printf 'absent-or-not-regular %s\n' "$RELATIVE" >> "$OUTPUT/observations.txt"
  fi
}

record_receipt_digest() {
  RELATIVE=$1
  if ssh -o BatchMode=yes -o ConnectTimeout="$SSH_CONNECT_TIMEOUT_SECONDS" "$HOST" "test -f '$STAGING/$RELATIVE' && test ! -L '$STAGING/$RELATIVE'"; then
    DIGEST_LINE=$(ssh -o BatchMode=yes -o ConnectTimeout="$SSH_CONNECT_TIMEOUT_SECONDS" "$HOST" "printf '%s ' \"\$(stat -c %s '$STAGING/$RELATIVE')\"; /home/brittonr/mantle-runs/receipt-fix-v31/b3sum-1.8.5-operator --no-names '$STAGING/$RELATIVE'")
    printf 'receipt %s bytes_blake3=%s\n' "$RELATIVE" "$DIGEST_LINE" >> "$OUTPUT/observations.txt"
  else
    OBSERVATION_STATUS=$?
    EXPECTED_ABSENCE_STATUS=1
    test "$OBSERVATION_STATUS" -eq "$EXPECTED_ABSENCE_STATUS"
    printf 'receipt absent-or-not-regular %s\n' "$RELATIVE" >> "$OUTPUT/observations.txt"
  fi
}
