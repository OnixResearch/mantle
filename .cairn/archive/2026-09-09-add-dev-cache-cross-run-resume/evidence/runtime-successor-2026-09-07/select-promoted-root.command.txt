set -eu
RUN=$1
NAME=$2
STATUS="$RUN/evidence/$NAME.status.txt"
test -d "$RUN"
test ! -L "$RUN"
test -f "$STATUS"
test ! -L "$STATUS"
grep -q '^finished_at=' "$STATUS"
grep -Eq '^exit_code=[0-9]+$' "$STATUS"
ROOT_COUNT=0
SELECTED_ROOT=
for CANDIDATE in "$RUN/$NAME" "$RUN/.$NAME.source-built-fixed-point-staging-"*; do
  if test -e "$CANDIDATE" || test -L "$CANDIDATE"; then
    test -d "$CANDIDATE"
    test ! -L "$CANDIDATE"
    ROOT_COUNT=$((ROOT_COUNT + 1))
    SELECTED_ROOT=$CANDIDATE
  fi
done
test "$ROOT_COUNT" -eq 1
printf '%s\n' "$SELECTED_ROOT"
